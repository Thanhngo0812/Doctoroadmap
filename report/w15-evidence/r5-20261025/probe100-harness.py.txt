location=root/'tmp/w15-r5-probe100-location.json'
assert not location.exists()
bundle=json.loads((root/'tmp/w15-r5-authorized-fa80b31ad6b6.json').read_text(encoding='utf-8'))
worst=Decimal(0)
for item in bundle['attempts']:
    result=json.loads((root/item['run_dir']/'execution.json').read_text(encoding='utf-8'))
    receipt=result.get('bootstrap') or result['attempts'][0]['result']
    budget=receipt['step_details']['operator_budget']
    worst+=Decimal(budget['known_usd'])+Decimal(budget['unknown_upper_usd'])
for name in ('diagnostic','controlled'):
    loc=json.loads((root/('tmp/w15-r5-'+name+'-location.json')).read_text())
    if name=='diagnostic':
        result=json.loads((root/loc['run_dir']/'diagnostic-result.json').read_text(encoding='utf-8'))
        budget=result['approval_record']['budget']
    else:
        # Do not add traffic to the independent eight-worker comparison while it is running.

        budget=result['attempts'][0]['result']['step_details']['operator_budget']
    worst+=Decimal(budget['known_usd'])+Decimal(budget['unknown_upper_usd'])
assert worst+Decimal('.40')<=Decimal(4)
original_profile=operator.profile
def profile(run_dir,database):
    settings,core=original_profile(run_dir,database)
    return settings,core.model_copy(update={'llm_max_in_flight':100,'gemini_max_retries':0})
operator.profile=profile
plan_path=operator.prepare('candidate')
plan=json.loads(plan_path.read_text(encoding='utf-8'))
run=plan_path.parent
settings,core=profile(run,plan['database'])
engine=operator.create_database_engine(operator.test_database_url(plan['database']))
runtime=RealBuildRuntime(settings,engine,core_settings=core)
assert runtime.pipeline_version==plan['pipeline_version']
reference='w15-owner-probe100-20261025-'+uuid4().hex
permission={key:plan[key] for key in ('file_sha256','pipeline_version','database','embedding_price_profile')}
permission.update(approved=True,approved_by='project-owner',reference=reference,max_usd='0.40',
    embedding_billing_verified=False,owner_debug_quote='nếu cần chi tiền tìm lỗi thì cứ chi',
    owner_concurrency_quote='ok cho phép bạn tăng lên 100 test đấy nhưng sau giờ nghỉ trưa, tạm ngưng công việc chút',
    owner_resume_quote='tiếp tục công việc',operation='isolated-100-embedding-and-one-generation-control',
    max_provider_retries=0,max_in_flight=100,plan_sha256=sha256(plan_path.read_bytes()).hexdigest(),
    previous_attempts_upper_usd=str(worst),upper_including_this_cap_usd=str(worst+Decimal('.40')))
approval_path=run/'owner-probe100-approval.json'
approval_path.write_bytes((json.dumps(permission,ensure_ascii=False,indent=2)+'\n').encode())
location.write_bytes((json.dumps(dict(run_dir=run.relative_to(root).as_posix(),database=plan['database'],
    reference=reference,cap_usd='0.40',plan_sha256=sha256(plan_path.read_bytes()).hexdigest(),
    approval_sha256=sha256(approval_path.read_bytes()).hexdigest()),indent=2)+'\n').encode())
grant=build_spend.BuildSpendApproval(**{k:permission[k] for k in ('reference','max_usd','file_sha256','pipeline_version','embedding_price_profile')},audit_log=run/'budget-requests.jsonl')
lock=Lock()
active={'current':0,'peak':0,'dispatch':0}
real_urlopen=build_spend.urlopen
def observed_provider(request,*args,**kwargs):
    with lock:
        active['current']+=1;active['dispatch']+=1
        active['peak']=max(active['peak'],active['current'])
        sequence=active['dispatch']
    row=dict(sequence=sequence,operation=request.full_url.rsplit(':',1)[-1],
        started_at=datetime.now().astimezone().isoformat(),timeout_seconds=kwargs.get('timeout'),
        request_sha256=sha256(request.data).hexdigest(),http_status=None)
    start=perf_counter()
    try:
        with real_urlopen(request,*args,**kwargs) as response:
            row['http_status']=response.status
            raw=response.read()
        value=json.loads(raw)
        row.update(response_keys=list(value),usageMetadata=value.get('usageMetadata'),
            embedding_statistics=[p.get('embeddings',{}).get('statistics') for p in value.get('predictions',[])])
        return BytesIO(raw)
    except Exception as error:
        row.update(error_type=type(error).__name__,error_message=str(error)[:600],
            reason_type=type(getattr(error,'reason',None)).__name__,http_status=getattr(error,'code',row['http_status']),
            traceback=traceback.format_exc())
        raise
    finally:
        row['wall_seconds']=perf_counter()-start
        with lock:
            active['current']-=1
            with (run/'transport-trace.jsonl').open('a',encoding='utf-8') as handle:
                handle.write(json.dumps(row,ensure_ascii=False)+'\n')

source_item=next(item for item in bundle['attempts'] if item['book']=='candidate')
store=JsonArtifactStore(root/source_item['run_dir']/'artifacts')
chunks=sorted(store.load_latest_chunks('u-'+plan['file_sha256'][:20]).result.chunks,key=lambda c:c.order_index)[:100]
assert len(chunks)==100
inputs=[EmbeddingRequest(embedding_text(chunk.section_path,chunk.text),'RETRIEVAL_DOCUMENT') for chunk in chunks]
input_manifest=[dict(chunk_id=c.chunk_id,text_sha256=sha256(r.text.encode()).hexdigest(),characters=len(r.text)) for c,r in zip(chunks,inputs)]
(run/'input-hashes.json').write_bytes((json.dumps(input_manifest,indent=2)+'\n').encode())
outcome=dict(operation=permission['operation'],configured_concurrency=100,requested_embeddings=100,
    generation_control_requests=1,actual_generation_100_not_tested=True,embedding_billing_verified=False)
pipeline=None
try:
    with patch.object(build_spend,'urlopen',observed_provider),grant.activate(run/'source.pdf','u-'+plan['file_sha256'][:20],runtime) as guard:
        pipeline=runtime.pipeline_factory(core,allow_spend=True,phase_b_store=runtime.phase_b_store)
        guard.stage='generation_control'
        model=MeteredModel(pipeline._extractor(core.extraction_model_id,max_output_tokens=32),pipeline.usage_meter)
        reply=model.start_session('Return only the requested JSON object.').ask_json('Return {"ok":true}.',
            response_schema={'type':'object','properties':{'ok':{'type':'boolean'}},'required':['ok']})
        outcome['generation_ok']=reply.payload.get('ok')
        # Refresh the shared embedding client before fanout, without logging its credentials.
        pipeline.embedding._vertex.headers()
        guard.stage='embedding_100'
        start=perf_counter()
        vectors=pipeline.embedding_guard.embed(inputs)
        outcome.update(embedding_wall_seconds=perf_counter()-start,embedding_results=len(vectors),
            all_dimensions_768=all(len(v)==768 for v in vectors),status='diagnostic_succeeded')
except Exception as error:
    outcome.update(status='diagnostic_failed',error_type=type(error).__name__,error_message=str(error)[:600],traceback=traceback.format_exc())
finally:
    usage=pipeline.usage() if pipeline else dict(spent_usd=0,model_calls=0,input_tokens=0,output_tokens=0,usage_complete=True)
    record=runtime.phase_b_store.get_record('w15_build_approval',reference)
    outcome.update(usage=usage,approval_record=record,transport_counts=active,
        limiter=pipeline.limiter.snapshot() if pipeline else None)
    runtime.phase_b_store.put_record('w15_debug_receipt',reference,outcome)
    runtime.phase_b_store.append_usage(dict(purpose='build',doc_id='u-'+plan['file_sha256'][:20],
        operation=permission['operation'],approval_reference=reference,cost_usd=round(usage['spent_usd'],6),
        input_tokens=usage['input_tokens'],output_tokens=usage['output_tokens'],model_calls=usage['model_calls'],
        usage_complete=usage['usage_complete'],embedding_billing_verified=False))
    (run/'diagnostic-result.json').write_bytes((json.dumps(outcome,ensure_ascii=False,indent=2,default=str)+'\n').encode())
    runtime.close();engine.dispose()
    print(json.dumps(outcome,ensure_ascii=False,indent=2,default=str),flush=True)
