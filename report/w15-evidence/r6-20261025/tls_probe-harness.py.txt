"""One TLS-only connection to the existing Vertex host; no HTTP/model request or credentials."""
from datetime import datetime
import json
from pathlib import Path
import socket
import ssl
from time import perf_counter
root=Path(__file__).resolve().parents[1]
host='aiplatform.googleapis.com'
start=perf_counter()
value=dict(host=host,checked_at=datetime.now().astimezone().isoformat(),http_requests=0,provider_calls=0)
try:
    value['dns_addresses']=sorted({r[4][0] for r in socket.getaddrinfo(host,443,type=socket.SOCK_STREAM)})
    with socket.create_connection((host,443),timeout=10) as tcp:
        value.update(tcp_peer=tcp.getpeername(),dns_and_connect_seconds=perf_counter()-start)
        tls_start=perf_counter()
        with ssl.create_default_context().wrap_socket(tcp,server_hostname=host) as tls:
            cert=tls.getpeercert()
            value.update(tls_seconds=perf_counter()-tls_start,tls_version=tls.version(),cipher=tls.cipher(),
                certificate_issuer=cert.get('issuer'),certificate_subject=cert.get('subject'),
                certificate_not_before=cert.get('notBefore'),certificate_not_after=cert.get('notAfter'),certificate_verified=True)
except Exception as error:
    value.update(error_type=type(error).__name__,error_message=str(error))
value['seconds']=perf_counter()-start
(root/'report/w15-evidence/r6-20261025/tls-public-probe.json').write_bytes((json.dumps(value,indent=2)+'\n').encode())
print(json.dumps(value,indent=2))
