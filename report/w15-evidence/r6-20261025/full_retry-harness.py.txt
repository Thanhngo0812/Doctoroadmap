real_urlopen=build_spend.urlopen

state=local();lock=Lock();counts=dict(sequence=0,active=0,peak=0)
def create_connection(*args,**kwargs):
    start=perf_counter()
    sock=real_create_connection(*args,**kwargs)
    row=getattr(state,'row',None)
    if row is not None:row.update(tcp_peer=sock.getpeername(),tcp_seconds=perf_counter()-start)
    return sock
def wrap_socket(context,*args,**kwargs):
    start=perf_counter()
    sock=real_wrap_socket(context,*args,**kwargs)
    row=getattr(state,'row',None)
    if row is not None:row.update(tls_handshake_seconds=perf_counter()-start,tls_handshake_finished=True)
    return sock
def observe_guard(guard,*args,**kwargs):
    state.stage=guard.stage
    return real_guard_call(guard,*args,**kwargs)
def socket_read(sock,buffer):
    row=getattr(state,'row',None)
    if row is None:return real_readinto(sock,buffer)
    start=perf_counter();event=dict(phase=row['phase'],requested_bytes=len(buffer))
    try:
        n=real_readinto(sock,buffer)
        event['read_bytes']=n
        row['wire_bytes_read']+=n or 0
        return n
    except Exception as error:
        event['error_type']=type(error).__name__
        raise
    finally:
        event['seconds']=perf_counter()-start
        row['socket_reads'].append(event)
def observed_provider(request,*args,**kwargs):
    payload=json.loads(request.data)
    with lock:
        counts['sequence']+=1;counts['active']+=1;counts['peak']=max(counts['peak'],counts['active'])
        sequence=counts['sequence']
    row=dict(sequence=sequence,stage=getattr(state,'stage',None),operation=request.full_url.rsplit(':',1)[-1],
        request_sha256=sha256(request.data).hexdigest(),request_bytes=len(request.data),
        task_type=payload.get('instances',[{}])[0].get('task_type'),
        started_at=datetime.now().astimezone().isoformat(),timeout_seconds=kwargs.get('timeout'),
        phase='urlopen',http_status=None,wire_bytes_read=0,socket_reads=[])
    state.row=row;start=perf_counter();response=None
    with lock:
        with (run/'dispatch.jsonl').open('a',encoding='utf-8') as f:
            f.write(json.dumps({k:row[k] for k in ('sequence','stage','operation','task_type','request_sha256','started_at')})+'\n')
    try:
        with real_urlopen(request,*args,**kwargs) as response:
            row.update(http_status=response.status,headers_seconds=perf_counter()-start,
                headers={k.lower():v for k,v in response.headers.items() if k.lower() in
                    {'content-type','content-length','content-encoding','transfer-encoding','connection','server','date','x-request-id','x-goog-request-id'}},
                http_version=response.version,chunked=response.chunked,will_close=response.will_close,
                length_at_headers=response.length,wire_bytes_at_headers=row['wire_bytes_read'])
            try:
                sock=response.fp.raw._sock
                row['tls']=dict(version=sock.version(),cipher=sock.cipher(),alpn=sock.selected_alpn_protocol(),peer=sock.getpeername(),timeout=sock.gettimeout())
            except (AttributeError,OSError):
                row['tls']=None
            row['phase']='body'
            raw=response.read()
            row.update(body_bytes=len(raw),body_sha256=sha256(raw).hexdigest(),body_seconds=perf_counter()-start-row['headers_seconds'])
        value=json.loads(raw)
        row.update(usageMetadata=value.get('usageMetadata'),embedding_statistics=[p.get('embeddings',{}).get('statistics') for p in value.get('predictions',[])])
        return BytesIO(raw)
    except Exception as error:
        row.update(error_type=type(error).__name__,error_message=str(error)[:600],traceback=traceback.format_exc(),
            http_status=getattr(error,'code',row['http_status']),length_after_error=getattr(response,'length',None),
            chunk_left_after_error=getattr(response,'chunk_left',None))
        raise
    finally:
        row['wall_seconds']=perf_counter()-start
        state.row=None
        with lock:
            counts['active']-=1
            with (run/'transport-trace.jsonl').open('a',encoding='utf-8') as f:f.write(json.dumps(row,ensure_ascii=False)+'\n')
