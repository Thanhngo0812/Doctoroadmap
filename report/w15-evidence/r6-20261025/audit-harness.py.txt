"""Zero-provider SQL/artifact/accounting checks over the finished core-retry book."""
from collections import Counter,defaultdict
from datetime import datetime
from decimal import Decimal
from hashlib import sha256
import json
from pathlib import Path
import sys
root=Path(__file__).resolve().parents[1]
sys.path[:0]=[str(root/'src'),str(root/'src/web/backend'),str(root/'src/web/backend/tests')]
import w15_operator_run as operator
from sqlalchemy import text
from PIL import Image
from doc2roadmap.infrastructure.artifacts.json_store import JsonArtifactStore
from doc2roadmap.application.phase_b.retrieval import load_learning_graph,assert_retrieval_compatible
from doc2roadmap.domain.global_validation import strongly_connected_components
out=root/'report/w15-evidence/r6-20261025'
item=json.loads((root/'tmp/w15-r6-full-retry-location.json').read_text());run=root/item['run_dir']
plan=json.loads((run/'plan.json').read_text(encoding='utf-8'))
execution=json.loads((run/'execution.json').read_text(encoding='utf-8'))
receipt=execution['attempts'][0]['result'];budget=receipt['step_details']['operator_budget']
assert receipt['status']=='succeeded' and execution['mail_subjects']==['Bản dựng xong']
assert receipt['pipeline_version']==plan['pipeline_version'] and receipt['doc_id']=='u-'+plan['file_sha256'][:20]
assert sha256((run/'source.pdf').read_bytes()).hexdigest()==plan['file_sha256'] and (run/'source.pdf').stat().st_nlink==1
events=[]
for p in run.glob('requests-*.jsonl'):events += [json.loads(l) for l in p.read_text(encoding='utf-8').splitlines()]
traces=[json.loads(l) for l in (run/'transport-trace.jsonl').read_text(encoding='utf-8').splitlines()]
known=sum((Decimal(e['actual_usd']) for e in events if e['actual_usd'] is not None),Decimal(0))
unknown=sum((Decimal(e['reserved_usd']) for e in events if e['actual_usd'] is None),Decimal(0))
assert len(events)==len(traces)==budget['http_calls'] and budget['held_usd']=='0'
assert known==Decimal(budget['known_usd']) and unknown==Decimal(budget['unknown_upper_usd'])
assert known+unknown<=Decimal(budget['limit_usd']) and budget['retry_unknown_within_cap'] is True
engine=operator.create_database_engine(operator.test_database_url(item['database']))
try:
    with engine.connect() as c:
        approval=c.execute(text("SELECT value FROM phase_b_record WHERE kind='w15_build_approval' AND record_id=:ref"),{'ref':item['reference']}).scalar_one()
        builds=[dict(r) for r in c.execute(text('SELECT id,status,doc_id,pipeline_version,spent_usd,model_calls,usage_complete,claim_count FROM build')).mappings()]
        courses=[dict(r) for r in c.execute(text('SELECT id,status,doc_id,published_graph_version FROM course')).mappings()]
        ledger=[dict(r) for r in c.execute(text('SELECT purpose,cost_usd,input_tokens,output_tokens,extra FROM usage_ledger ORDER BY id')).mappings()]
        attempt_count=c.execute(text('SELECT count(*) FROM build_attempt')).scalar_one()
    assert approval['state']=='finished' and approval['retry_unknown_within_cap'] is True and approval['budget']==budget
    assert approval['events']==events
    assert approval['file_sha256']==plan['file_sha256'] and approval['pipeline_version']==plan['pipeline_version']
    assert len(builds)==len(courses)==attempt_count==1 and builds[0]['claim_count']==1
    assert builds[0]['status']=='succeeded' and courses[0]['status']=='ready'
    assert sum((r['cost_usd'] for r in ledger),Decimal(0))==Decimal(str(receipt['spent_usd']))
    assert all(r['cost_usd']==r['input_tokens']==r['output_tokens']==0 for r in ledger if r['extra'].get('counted_in')=='build')
finally:engine.dispose()
store=JsonArtifactStore(run/'artifacts');graph=store.load_graph(receipt['doc_id'],receipt['graph_version'])
lg=load_learning_graph(store,receipt['doc_id'],graph.version);ids=set(lg.learnable_nodes)
assert courses[0]['published_graph_version']==graph.version
members=[n for m in graph.payload['modules'] for n in m['node_ids']]
assert sorted(members)==sorted(ids) and sorted(graph.payload['sequence'])==sorted(ids)
hard=[e for e in [*graph.payload['requires_edges'],*graph.payload.get('bypassed_edges',[])] if not e.get('soft') and e['source'] in ids and e['target'] in ids]
assert all(len(p)==1 for p in strongly_connected_components(ids,hard))
ranks={n:i for i,n in enumerate(graph.payload['sequence'])}
assert all(ranks[e['source']]<ranks[e['target']] for e in hard)
chunks=store.load_chunks(receipt['doc_id'],graph.manifest['config']['chunks_version'])
fingerprint=assert_retrieval_compatible(store.load_latest_retrieval(receipt['doc_id']).payload,chunks.result.chunks)
images=[]
for p in sorted((run/'artifacts').rglob('*')):
    if p.suffix.lower() in ('.jpg','.jpeg','.png','.webp'):
        with Image.open(p) as im:im.verify()
        with Image.open(p) as im:images.append(dict(path=p.relative_to(run).as_posix(),size=list(im.size),format=im.format,sha256=sha256(p.read_bytes()).hexdigest()))
peaks={}
for stage in sorted({r['stage'] for r in traces}):
    points=[]
    for r in traces:
        if r['stage']!=stage:continue
        start=datetime.fromisoformat(r['started_at']).timestamp();points.extend([(start,1),(start+r['wall_seconds'],-1)])
    active=peak=0
    for _,d in sorted(points):active+=d;peak=max(peak,active)
    peaks[stage]=peak
chains=defaultdict(list)
for r in traces:chains[r['request_sha256']].append(r)
recovered=[]
for h,rs in chains.items():
    errors=[r for r in rs if r.get('error_type')]
    if errors:
        successes=[r for r in rs if not r.get('error_type') and r['sequence']>min(e['sequence'] for e in errors)]
        recovered.append(dict(request_sha256=h,errors=[{k:r.get(k) for k in ('sequence','stage','operation','http_status','error_type','phase','wall_seconds')} for r in errors],
            later_successes=[{k:r.get(k) for k in ('sequence','stage','operation','http_status','wall_seconds')} for r in successes]))
previous=json.loads((root/'report/w15-evidence/r5-20261025/summary.json').read_text())
combined_known=Decimal(previous['known_demo_usd'])+known;combined_unknown=Decimal(previous['unknown_upper_usd'])+unknown
assert combined_known+combined_unknown<=Decimal(4)
value=dict(location=item,plan=plan,receipt=receipt,budget=budget,sql=dict(builds=builds,courses=courses,attempt_count=attempt_count,ledger=ledger,
    approval={k:v for k,v in approval.items() if k!='events'},approval_event_count=len(events),approval_events_match_request_log=True),
    graph=dict(version=graph.version,module_count=len(graph.payload['modules']),learnable_count=len(ids),mentioned_count=len(graph.payload.get('mentioned_nodes',[])),
        non_learnable_count=len(graph.payload.get('non_learnable_nodes',[])),edges={k:len(v) for k,v in graph.payload.items() if k.endswith('edges')},
        quality_gate=graph.payload.get('quality_gate'),module_titles=[m['title'] for m in graph.payload['modules']],formula_refs=len(lg.formula_images),
        retrieval_input_fingerprint=fingerprint,graph_manifest=graph.manifest),artifact_images=images,
    actual_http_calls=len(traces),http_status_counts=dict(Counter(str(r.get('http_status')) for r in traces)),stage_peaks=peaks,recovered_error_chains=recovered,
    generation_100_simultaneously_proven=False,embedding_timeout_reproduced=any(r.get('error_type')=='TimeoutError' for r in traces),
    core_unchanged=True,usage_complete=receipt['usage_complete'],new_known_demo_usd=str(known),new_unknown_upper_usd=str(unknown),
    combined_known_demo_usd=str(combined_known),combined_unknown_upper_usd=str(combined_unknown),combined_upper_usd=str(combined_known+combined_unknown),
    invoice_reconciled=False,embedding_billing_verified=False,
    assertions=dict(sql_ready_and_exactly_one_attempt=True,pinned_graph_identity=True,ledger_equals_meter_delta=True,
        raw_trace_not_double_billed=True,all_learnable_covered_once=True,hard_requires_dag_and_topological_sequence=True,
        retrieval_inputs_compatible=True,ordinary_source_copy_sha=True,unknown_reserved_inside_cap=True))
(out/'final-audit.json').write_bytes((json.dumps(value,ensure_ascii=False,indent=2,default=str)+'\n').encode())
print(json.dumps({k:v for k,v in value.items() if k in ('new_known_demo_usd','new_unknown_upper_usd','combined_upper_usd','stage_peaks','http_status_counts','usage_complete','assertions')},ensure_ascii=False,indent=2))
print(json.dumps({k:v for k,v in value['graph'].items() if k in ('version','module_count','learnable_count','formula_refs','quality_gate')},ensure_ascii=False,indent=2))
