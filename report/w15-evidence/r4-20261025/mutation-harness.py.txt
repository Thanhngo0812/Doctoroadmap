"""Three falsifiable W15 R4 regressions; restore each exact source byte sequence."""
from hashlib import sha256
import json
import os
from pathlib import Path
import subprocess
import sys
import xml.etree.ElementTree as ET

root = Path(__file__).resolve().parents[1]
output = root / 'report/w15-evidence/mutations/r4-20261025'
output.mkdir(parents=True, exist_ok=True)
cases = [
    ('portable-plan', 'src/web/backend/tests/w15_operator_run.py',
     'run_dir=run_dir.relative_to(ROOT).as_posix()', 'run_dir=str(run_dir.relative_to(ROOT))',
     'src/web/backend/tests/test_operator_w15.py', 'prepare_saved_plan'),
    ('unverified-embedding', 'src/web/backend/wayfinder/infrastructure/build_spend.py',
     "if url.path.endswith('/models/gemini-embedding-001:predict'):\n            require_embedding_price_profile()",
     "if url.path.endswith('/models/gemini-embedding-001:predict'):",
     'src/web/backend/tests/test_build_spend_w15.py', 'unverified_online'),
    ('unverified-grant', 'src/web/backend/wayfinder/infrastructure/build_spend.py',
     '        require_embedding_price_profile()\n        guard = BuildSpendGuard',
     '        guard = BuildSpendGuard', 'src/web/backend/tests/test_build_approval_w15.py', 'unverified_online'),
]
records = []
for label, relative, old, new, test, selection in cases:
    path = root / relative
    original = path.read_bytes()
    source = original.decode()
    assert source.count(old) == 1
    xml = output / (label + '.xml')
    try:
        path.write_bytes(source.replace(old, new).encode())
        command = [sys.executable, 'scripts/w05_run_tests.py', test, '-k', selection,
                   '-o', 'addopts=', '-q', '--tb=short', '--basetemp=tmp/w15-r4-mut-' + label,
                   '--junitxml=' + str(xml)]
        process = subprocess.run(command, cwd=root, capture_output=True, encoding='utf-8', errors='replace',
            env={**os.environ, 'PYTHONDONTWRITEBYTECODE': '1', 'W15_EVIDENCE_RUN': 'r4-mutation-' + label,
                 'W25_EVIDENCE_RUN': 'w15-r4-mutation-' + label})
        (output / (label + '.log')).write_bytes((process.stdout + process.stderr).encode())
        tree = ET.parse(xml)
        failures, errors = tree.findall('.//testcase/failure'), tree.findall('.//testcase/error')
        assertions = sum(any(word in (r.text or '') for word in ('AssertionError', 'assert ', 'DID NOT RAISE',
                         'Failed: prepare output was rejected before composition')) for r in failures)
        assert process.returncode == 1 and assertions and not errors, (label, process.stdout[-1000:])
    finally:
        path.write_bytes(original)
        assert sha256(path.read_bytes()).hexdigest() == sha256(original).hexdigest()
    records.append(dict(label=label, source=relative, original_sha256=sha256(original).hexdigest(),
                        failures=len(failures), assertion_failures=assertions, setup_errors=len(errors),
                        native_exit=process.returncode, source_restored=True, command=command))
    (output / 'mutations.json').write_bytes((json.dumps(records, indent=2) + '\n').encode())
    print(label + ': assertion caught; source restored', flush=True)
