"""Ordinary verified fixture copies for the existing zero-spend full gate only."""
from hashlib import sha256
import json
from pathlib import Path
import shutil
import sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'src/web/backend'))
from wayfinder.domain.file_paths import safe_path

entry = json.loads((root / 'report/w15-evidence/candidate.json').read_text(encoding='utf-8'))
assert entry['candidate'] == 'document-data/train/nong-nghiep/rau-thuy-canh.pdf'
assert entry['staged_copy'] == 'tmp/w15-candidate/input.pdf'
source = safe_path(root.parents[1], entry['candidate'])
target = safe_path(root, entry['staged_copy'])
assert source.stat().st_nlink == 1
assert sha256(source.read_bytes()).hexdigest() == entry['sha256']
assert not target.exists()
target.parent.mkdir(parents=True, exist_ok=True)
shutil.copyfile(source, target)
assert target.stat().st_nlink == 1 and sha256(target.read_bytes()).hexdigest() == entry['sha256']
proof = dict(source=entry['candidate'], target=entry['staged_copy'], sha256=entry['sha256'],
    bytes=target.stat().st_size, ordinary_copy=True,
    purpose='Existing full-gate fixture only, no W15 operator/paid replay or private run/database/cache copied')
out = root / 'report/w04-evidence'
(out / 'candidate-fixture-copy.json').write_text(json.dumps(proof, indent=2)+'\n', encoding='utf-8')
shutil.copyfile(root / 'report/w06-evidence/fixture-copy.json', out / 'development-fixture-copy.json')
print(json.dumps(proof))
