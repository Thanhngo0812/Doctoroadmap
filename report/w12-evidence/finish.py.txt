write(out/'retained-test-exports.json',dict(note='Original generated test exports retained unchanged outside W12 commit',

helpers=[]

changed=git('diff','--name-only',main,'HEAD').splitlines()
assert all(name.startswith('src/web/backend/') or name=='src/web/openapi.json' for name in changed)

write(out/'source-identity.json',dict(head_tested=head,main_received=main,changed_files=source_identity,
    note='Worktree bytes used by tests and Git blob bytes both hashed; line endings can differ on Windows'))

values=dotenv_values(root.parents[1]/'src/web/.env')
secrets=[value for key,value in values.items() if value and len(value)>=8 and
         any(part in key.lower() for part in ('password','secret','token','api_key'))]
raw=root/'tmp/w12-raw-evidence';entries=[];redactions=[]
paths=[report,*sorted(p for p in out.rglob('*') if p.is_file())]
for path in paths:
    before=path.read_bytes()
    relative=path.relative_to(root)
    backup=raw/relative
    backup.parent.mkdir(parents=True,exist_ok=True)
    assert not backup.exists()
    backup.write_bytes(before)
    value=before.decode('utf-8-sig').replace('\r\n','\n')
    for secret in secrets:
        if secret in value:
            redactions.append(dict(path=relative.as_posix(),kind='configured secret',occurrences=value.count(secret)))
            value=value.replace(secret,'CONFIG_SECRET_OMITTED')
    # A traceback may contain only ephemeral synthetic email token values; do not publish those either.
    value,n=re.subn(r'(#(?:invitation|token)=)[A-Za-z0-9_-]{43}(?![A-Za-z0-9_-])',r'\1FIXTURE_TOKEN_OMITTED',value)
    if n:redactions.append(dict(path=relative.as_posix(),kind='synthetic URL fragment token',occurrences=n))
    normalized='\n'.join(line.rstrip(' \t') for line in value.split('\n')).encode()
    if path.suffix=='.xml':
        before_nodes=list(ET.fromstring(before).iter());after_nodes=list(ET.fromstring(normalized).iter())
        assert len(before_nodes)==len(after_nodes)
        for original,current in zip(before_nodes,after_nodes):
            assert original.tag==current.tag
            omit={'message'} if original.tag in {'failure','error','skipped'} else set()
            assert {k:v for k,v in original.attrib.items() if k not in omit}=={k:v for k,v in current.attrib.items() if k not in omit}
    assert not any(secret.encode() in normalized for secret in secrets)
    path.write_bytes(normalized)
    if path!=report:
        entries.append(dict(path=path.relative_to(out).as_posix(),raw_sha256=sha256(before).hexdigest(),
                            sha256=sha256(normalized).hexdigest(),bytes=len(normalized)))
write(out/'redactions.json',dict(items=redactions,unredacted_originals_retained_in_ignored_tmp=True))
redacted=out/'redactions.json';data=redacted.read_bytes()
entries.append(dict(path=redacted.name,raw_sha256=sha256(data).hexdigest(),sha256=sha256(data).hexdigest(),bytes=len(data)))
write(out/'manifest.json',dict(files=entries))
