def diagnosed_provider(*args, **kwargs):
    try:
        return urlopen(*args, **kwargs)
    except HTTPError as error:
        try:
            body = json.loads(error.read()).get('error', {})
            diagnostic = {'status': body.get('status'), 'message': body.get('message', '')[:1200],
                          'code': error.code, 'retry_after': error.headers.get('Retry-After')}
        except Exception:
            diagnostic = {'code': error.code, 'body_unavailable': True}
        with (OUT / 'provider-errors.jsonl').open('a', encoding='utf-8') as stream:
            stream.write(json.dumps({'attempt': attempt, 'utc': datetime.now(timezone.utc).isoformat(), **diagnostic}) + '\n')
        raise

def safe_open(original):
    def guarded(file, *args, **kwargs):
        if not isinstance(file, int):
            name = str(file).replace('\\', '/').lower()
            if any(part in name for part in protected):
                raise PermissionError('Protected path: ' + name)
        return original(file, *args, **kwargs)
    return guarded

def phase_store(path, *args, **kwargs):
    if str(path).replace('\\', '/').endswith('phase_b/live.json'):
        path = scoped_store
    return original_store(path, *args, **kwargs)

def event(kind, **details):
    row = {'utc': datetime.now(timezone.utc).isoformat(), 'event': kind,
           'budget': guard.snapshot(), **details}
    with (OUT / 'paid-progress.jsonl').open('a', encoding='utf-8') as stream:
        stream.write(json.dumps(row, ensure_ascii=False) + '\n')
    print(json.dumps(row, ensure_ascii=False), flush=True)

def trace(pipeline, name):
    original = getattr(pipeline, name)
    @functools.wraps(original)
    def measured(*args, **kwargs):
        guard.stage = name
        started = perf_counter()
        event('stage_started', stage=name)
        try:
            value = original(*args, **kwargs)
            detail = value if name != 'read' else {k: v for k, v in value.items() if k != 'page_texts'}
            event('stage_completed', stage=name, seconds=perf_counter()-started, details=detail,
                  pipeline_usage=pipeline.usage())
            return value
        except Exception as error:
            event('stage_failed', stage=name, seconds=perf_counter()-started,
                  error_type=type(error).__name__, error=str(error))
            raise
    setattr(pipeline, name, measured)
